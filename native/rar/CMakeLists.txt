cmake_minimum_required(VERSION 3.25)
project(MirrorFluxRar C)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option ENABLE_MBEDTLS ENABLE_NETTLE ENABLE_OPENSSL ENABLE_MD ENABLE_LIBB2 ENABLE_LZ4 ENABLE_LZO ENABLE_LZMA ENABLE_ZSTD ENABLE_ZLIB ENABLE_BZip2 ENABLE_LIBXML2 ENABLE_BSDXML ENABLE_EXPAT ENABLE_WIN32_XMLLITE ENABLE_PCREPOSIX ENABLE_PCRE2POSIX ENABLE_LIBGCC ENABLE_CNG ENABLE_TAR ENABLE_CPIO ENABLE_CAT ENABLE_UNZIP ENABLE_XATTR ENABLE_ACL ENABLE_ICONV ENABLE_TEST ENABLE_INSTALL)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(POSIX_REGEX_LIB "NONE" CACHE STRING "" FORCE)
set(ENABLE_7Z OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_BZ2 OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_LZMA OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_ZLIB OFF CACHE BOOL "" FORCE)
set(BUILD_INTEGRATION_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(libarchive)
add_subdirectory(unarr)
add_library(mf_rar SHARED bridge.c rar15.c)
target_include_directories(mf_rar PRIVATE libarchive/libarchive unarr ${CMAKE_CURRENT_BINARY_DIR}/unarr)
target_compile_definitions(mf_rar PRIVATE LIBARCHIVE_STATIC)
target_link_libraries(mf_rar PRIVATE archive_static unarr)
if(MINGW)
  target_link_options(mf_rar PRIVATE -static -Wl,--no-insert-timestamp)
endif()
set_target_properties(mf_rar PROPERTIES PREFIX "")
